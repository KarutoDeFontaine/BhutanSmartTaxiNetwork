  <!-- APP JAVASCRIPT LOGIC -->
  <script>
    // 10 Station Data from physical exhibition model
    const STATIONS = [
      { id: 1, name: "Tashiling", distance: 1.0 },
      { id: 2, name: "Jampeling", distance: 1.5 },
      { id: 3, name: "Rabdeyling", distance: 2.0 },
      { id: 4, name: "Samdrupling", distance: 3.0 },
      { id: 5, name: "Pemaling", distance: 4.0 },
      { id: 6, name: "Lekithang", distance: 4.0 },
      { id: 7, name: "Choekhorling", distance: 5.0 },
      { id: 8, name: "Dzomlingthang", distance: 6.0 },
      { id: 9, name: "Dechenpelri", distance: 6.0 },
      { id: 10, name: "Pelrithang", distance: 7.0 }
    ];

    // Global State
    let currentUser = null; // { identifier, role }
    let walletBalance = 500; // Nu.
    let activeBooking = null; // Current ride details
    let driverEarnings = 0;
    let completedTrips = 0;

    // Initialize App
    window.onload = function() {
      calculateFare();
      renderStationsGrid();
    };

    // Authentication
    function handleLogin(e) {
      e.preventDefault();
      const input = document.getElementById('authInput').value.trim();
      const role = document.querySelector('input[name="role"]:checked').value;

      if (!input) {
        alert("Please enter a valid CID or Mobile Number!");
        return;
      }

      currentUser = { identifier: input, role: role };

      // Update UI Header
      document.getElementById('authSection').classList.add('hidden');
      document.getElementById('appDashboard').classList.remove('hidden');
      document.getElementById('userHeaderInfo').classList.remove('hidden');
      document.getElementById('userRoleBadge').innerText = role.toUpperCase();
      document.getElementById('userIdentifierDisplay').innerText = input;

      // Show specific view
      if (role === 'passenger') {
        document.getElementById('passengerView').classList.remove('hidden');
        document.getElementById('driverView').classList.add('hidden');
      } else {
        document.getElementById('driverView').classList.remove('hidden');
        document.getElementById('passengerView').classList.add('hidden');
        renderDriverQueue();
      }
    }

    function logout() {
      currentUser = null;
      document.getElementById('authSection').classList.remove('hidden');
      document.getElementById('appDashboard').classList.add('hidden');
      document.getElementById('userHeaderInfo').classList.add('hidden');
    }

    // Fare Calculation Logic
    function calculateFare() {
      const stationId = parseInt(document.getElementById('pickupPointSelect').value);
      const station = STATIONS.find(s => s.id === stationId);
      const baseFare = 20;
      const distanceFare = station.distance * 15; // Nu. 15 per km
      const total = baseFare + distanceFare;

      document.getElementById('distanceFareDisplay').innerText = `Nu. ${distanceFare.toFixed(2)}`;
      document.getElementById('totalFareDisplay').innerText = `Nu. ${total.toFixed(2)}`;
    }

    // Book Cab Request
    function requestCab() {
      const stationId = parseInt(document.getElementById('pickupPointSelect').value);
      const passengerCount = parseInt(document.getElementById('passengerCountSelect').value);
      const station = STATIONS.find(s => s.id === stationId);
      const fare = 20 + (station.distance * 15);

      if (walletBalance < fare) {
        alert("Insufficient wallet balance! Please recharge your account first.");
        openRechargeModal();
        return;
      }

      activeBooking = {
        id: 'REQ-' + Math.floor(1000 + Math.random() * 9000),
        station: station,
        passengerCount: passengerCount,
        fare: fare,
        status: 'DISPATCHED', // DISPATCHED -> COMPLETED / CANCELLED
        timestamp: new Date().toLocaleTimeString()
      };

      updateBookingUI();
      renderDriverQueue();
    }

    // Update Passenger Booking Visuals
    function updateBookingUI() {
      const noBookingCard = document.getElementById('noActiveBookingCard');
      const activeCard = document.getElementById('activeBookingCard');

      if (!activeBooking) {
        noBookingCard.classList.remove('hidden');
        activeCard.classList.add('hidden');
        return;
      }

      noBookingCard.classList.add('hidden');
      activeCard.classList.remove('hidden');

      document.getElementById('bookingLocationText').innerText = `Station ${activeBooking.station.id}: ${activeBooking.station.name}`;
      document.getElementById('bookingFareText').innerText = `Nu. ${activeBooking.fare.toFixed(2)}`;
      
      const statusText = document.getElementById('bookingStatusText');
      const progressBar = document.getElementById('bookingProgressBar');

      if (activeBooking.status === 'DISPATCHED') {
        statusText.innerText = 'Cab Dispatched - En Route';
        progressBar.style.width = '66%';
      } else if (activeBooking.status === 'COMPLETED') {
        statusText.innerText = 'Trip Completed';
        progressBar.style.width = '100%';
      }
    }

    // Ride Cancellation Logic with 50% Penalty
    function cancelRide() {
      if (!activeBooking) return;

      const cancellationFee = activeBooking.fare * 0.50;
      
      const confirmCancel = confirm(
        `Are you sure you want to cancel?\n\n` +
        `Cancellation Penalty (50%): Nu. ${cancellationFee.toFixed(2)}\n` +
        `This amount will be deducted from your wallet balance.`
      );

      if (confirmCancel) {
        walletBalance -= cancellationFee;
        updateWalletDisplay();

        alert(
          `Ride Cancelled.\n` +
          `Nu. ${cancellationFee.toFixed(2)} penalty applied.\n` +
          `New Wallet Balance: Nu. ${walletBalance.toFixed(2)}`
        );

        activeBooking = null;
        updateBookingUI();
        renderDriverQueue();
      }
    }

    // Driver Control Logic
    function renderDriverQueue() {
      const container = document.getElementById('driverRequestList');
      const ledDisplay = document.getElementById('ledDisplayContent');

      if (!activeBooking || activeBooking.status === 'COMPLETED') {
        container.innerHTML = `
          <div class="bg-slate-900/60 rounded-xl p-8 text-center text-slate-500 text-sm border border-slate-700/50">
            <i class="fa-solid fa-check-circle text-2xl mb-2 text-slate-600"></i>
            <p>No pending alerts from waiting points.</p>
          </div>
        `;
        ledDisplay.innerHTML = `
          <span>NO CALL REQUESTS</span>
          <span class="text-xs text-slate-600 font-normal">STANDBY MODE</span>
        `;
        return;
      }

      // Render Active Alert in Driver Queue
      container.innerHTML = `
        <div class="bg-slate-900 border border-amber-500/50 rounded-xl p-4 flex flex-wrap justify-between items-center gap-4 shadow-lg">
          <div class="space-y-1">
            <div class="flex items-center gap-2">
              <span class="bg-amber-500 text-slate-900 text-xs font-bold px-2 py-0.5 rounded">STATION ${activeBooking.station.id}</span>
              <h4 class="font-bold text-white text-base">${activeBooking.station.name} (${activeBooking.station.distance} km)</h4>
            </div>
            <p class="text-xs text-slate-400">
              <i class="fa-solid fa-users text-amber-400 mr-1"></i> ${activeBooking.passengerCount} Passenger(s) Waiting
              <span class="mx-2">•</span>
              <i class="fa-solid fa-money-bill-wave text-emerald-400 mr-1"></i> Fare: Nu. ${activeBooking.fare.toFixed(2)}
            </p>
          </div>

          <button onclick="driverAcceptAndComplete()" class="bg-emerald-600 hover:bg-emerald-500 text-white font-bold px-4 py-2.5 rounded-xl text-xs transition flex items-center gap-2">
            <i class="fa-solid fa-taxi"></i> Accept & Complete Trip
          </button>
        </div>
      `;

      // Update Simulated LED Board
      ledDisplay.innerHTML = `
        <span class="text-amber-400 animate-pulse">CAB REQUEST</span>
        <span class="text-sm font-bold text-white">${activeBooking.station.name}</span>
        <span class="text-xs text-amber-300 font-normal">${activeBooking.passengerCount} PASSENGER(S)</span>
      `;
    }

    function driverAcceptAndComplete() {
      if (!activeBooking) return;

      // Deduct fare from passenger & reward driver
      walletBalance -= activeBooking.fare;
      driverEarnings += activeBooking.fare;
      completedTrips += 1;

      updateWalletDisplay();
      document.getElementById('driverEarningsDisplay').innerText = `Nu. ${driverEarnings.toFixed(2)}`;
      document.getElementById('completedTripsCount').innerText = completedTrips;

      alert(`Trip Completed successfully! Nu. ${activeBooking.fare.toFixed(2)} credited to driver wallet.`);

      activeBooking.status = 'COMPLETED';
      activeBooking = null;

      updateBookingUI();
      renderDriverQueue();
    }

    // Wallet Recharge Logic
    function openRechargeModal() {
      document.getElementById('rechargeModal').classList.remove('hidden');
    }

    function closeRechargeModal() {
      document.getElementById('rechargeModal').classList.add('hidden');
    }

    function setRechargeAmount(amt) {
      document.getElementById('rechargeInput').value = amt;
    }

    function processRecharge() {
      const amt = parseFloat(document.getElementById('rechargeInput').value);
      if (isNaN(amt) || amt <= 0) {
        alert("Please enter a valid amount!");
        return;
      }

      walletBalance += amt;
      updateWalletDisplay();
      closeRechargeModal();
      alert(`Wallet recharged successfully with Nu. ${amt.toFixed(2)}!`);
    }

    function updateWalletDisplay() {
      document.getElementById('walletBalanceDisplay').innerText = `Nu. ${walletBalance.toFixed(2)}`;
    }

    // Render Stations Grid Visual Representation
    function renderStationsGrid() {
      const grid = document.getElementById('stationsGridPassenger');

          <div class="text-xs font-bold text-slate-200 truncate">${s.name}</div>
          <div class="text-[10px] text-amber-400 font-mono mt-0.5">${s.distance} km</div>
        </div>
      `).join('');
    }
